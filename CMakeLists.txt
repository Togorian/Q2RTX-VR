cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(Q2RTXVR LANGUAGES NONE)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
  message(FATAL_ERROR "Use the Ninja generator from an x64 MSVC developer shell.")
endif()
option(Q2RTX_VERIFY_REFERENCE_INPUTS "Verify exact reference source and library hashes" OFF)
set(Q2RTX_DEPENDENCIES "" CACHE PATH "Directory containing the six compatible x64 static/import libraries")
# Only real compilation inputs are required by the ordinary build.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/recipes/build-plan.json" build_plan)
foreach(kind C RC)
  string(JSON count LENGTH "${build_plan}" ${kind})
  math(EXPR last "${count}-1")
  foreach(i RANGE 0 ${last})
    string(JSON source GET "${build_plan}" ${kind} ${i} source)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
      message(FATAL_ERROR "Missing source file: ${source}. See docs/BUILDING.md.")
    endif()
  endforeach()
endforeach()
if(Q2RTX_VERIFY_REFERENCE_INPUTS)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/recipes/source-manifest.json" manifest)
  string(JSON count LENGTH "${manifest}" files)
  math(EXPR last "${count}-1")
  foreach(i RANGE 0 ${last})
    string(JSON relative GET "${manifest}" files ${i} path)
    string(JSON expected GET "${manifest}" files ${i} sha256)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${relative}")
      message(FATAL_ERROR "Missing reference input: ${relative}")
    endif()
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "Reference input differs: ${relative}; disable Q2RTX_VERIFY_REFERENCE_INPUTS for normal development.")
    endif()
  endforeach()
endif()
