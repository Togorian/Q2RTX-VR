cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
project(Q2RTXConnectionChecks LANGUAGES C)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This fixture requires Windows x64 MSVC.")
endif()
find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${ROOT}/recipes/product-version.cmake")
execute_process(COMMAND "${Python3_EXECUTABLE}" -I -B -S "${CMAKE_CURRENT_LIST_DIR}/materialise.py"
  --source "${ROOT}" --output "${CMAKE_BINARY_DIR}/excerpts"
  COMMAND_ERROR_IS_FATAL ANY)
add_executable(connection_checks checks.c)
target_include_directories(connection_checks PRIVATE "${Q2RTX_VERSION_OUTPUT}" "${CMAKE_BINARY_DIR}/excerpts")
file(READ "${ROOT}/recipes/build-plan.json" plan)
string(JSON count LENGTH "${plan}" C)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON unit GET "${plan}" C ${i} unit)
  if(unit STREQUAL "engine_043" OR unit STREQUAL "engine_054" OR unit STREQUAL "engine_065")
    string(JSON src GET "${plan}" C ${i} source)
    target_sources(connection_checks PRIVATE "${ROOT}/${src}")
    string(JSON ni LENGTH "${plan}" C ${i} includes)
    math(EXPR ilast "${ni}-1")
    foreach(j RANGE 0 ${ilast})
      string(JSON inc GET "${plan}" C ${i} includes ${j})
      target_include_directories(connection_checks PRIVATE "${ROOT}/${inc}")
    endforeach()
    string(JSON nf LENGTH "${plan}" C ${i} flags)
    math(EXPR flast "${nf}-1")
    foreach(j RANGE 0 ${flast})
      string(JSON flag GET "${plan}" C ${i} flags ${j})
      target_compile_options(connection_checks PRIVATE "${flag}")
    endforeach()
  endif()
endforeach()
target_compile_options(connection_checks PRIVATE /Gy /Gw)
target_link_options(connection_checks PRIVATE /OPT:REF /INCREMENTAL:NO)
